class Solution {
public:
    bool isValid(string s) {
        stack<char>st;
        int i=0,n=s.size();
        while(i<n){
            if(s[i]=='('||s[i]=='{'||s[i]=='[')st.push((s[i]));
            else{
                if(st.empty())return 0;
                if(s[i]==')'&&st.top()!='(')return 0;
                else if(s[i]=='}'&&st.top()!='{')return 0;
                else if(s[i]==']'&&st.top()!='[')return 0;
                st.pop();
            }
            i++;
        }return (st.empty())?1:0;
    }
};
