class Solution {
public:
    int evalRPN(vector<string>& tokens) {
        stack<int>st;
        int ans=0;
        for(auto x:tokens){
            if(x!="+"&&x!="-"&&x!="*"&&x!="/"){
                int val=stoi(x);
                st.push(val);
            }else{
                int a=st.top();st.pop();int b=st.top();st.pop();
                if(x=="+")st.push(a+b);
                else if(x=="-")st.push(b-a);
                else if(x=="*")st.push(a*b);
                else st.push(b/a);
            }
        }return st.top();
    }
};
