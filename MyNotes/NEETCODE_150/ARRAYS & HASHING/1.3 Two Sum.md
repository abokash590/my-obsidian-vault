class Solution {
public:
    vector<int> twoSum(vector<int>& nums, int target) {
        set<int>st;
        for(auto x:nums)st.insert(x);
        int a=0,b=0,ii,jj;
        for(auto x:st){
            if(st.count(target-x)){
                a=x,b=target-x;
                break;
            }
        }
        for(int i=0;i<nums.size();i++){
            if(nums[i]==a){
                ii=i;break;
            }
        }
        int ok=0;
        for(int i=0;i<nums.size();i++){
            if(nums[i]==b){
                if(a!=b){
                    jj=i;break;
                }else if(!ok)ok=1;
                else if(ok){
                    jj=i;
                    break;
                }
            }
        }
        vector<int>ans;ans.push_back(min(ii,jj));ans.push_back(max(ii,jj));
        return ans;
    }
};
