class Solution {
public:
    string encode(vector<string>& strs) {
        string s;
        for(auto x:strs)s+=x+"68985672";
        return s;
    }
    vector<string> decode(string s) {
        int n=s.size();
        string x="68985672";
        int i=0;
        vector<string>ans;
        while(s.find(x,i)<n){
            string str;
            int l=s.find(x,i);
            for(i;i<l;i++)str+=s[i];
            ans.emplace_back(str);
            i=l+8;
        }return ans;
    }
};
