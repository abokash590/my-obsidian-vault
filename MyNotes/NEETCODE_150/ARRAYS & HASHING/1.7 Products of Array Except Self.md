class Solution {
public:
    vector<int> productExceptSelf(vector<int>& nums) {
        int mul=1,cnt=0;
        for(auto x:nums)mul*=x;
        for(auto x:nums)if(!x)cnt++;
        vector<int>ans(nums.size());
        if(cnt>1)return ans;
        else if(cnt==1){
            int m=1;
            for(auto x:nums)if(x)m*=x;
            for(int i=0;i<nums.size();i++)if(!nums[i])ans[i]=m;
            return ans;
        }
        for(int i=0;i<nums.size();i++)ans[i]=mul/nums[i];
        return ans;
    }
};
