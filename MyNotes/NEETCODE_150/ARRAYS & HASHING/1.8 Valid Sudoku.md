class Solution {
public:
    bool isValidSudoku(vector<vector<char>>& board) {
        for(int i=0;i<9;i++){
            set<char>st;
            int cr=0;
            for(int j=0;j<9;j++){
                if(board[i][j]!='.')st.insert(board[i][j]),cr++;
            }if(st.size()!=cr)return 0;
            st.clear();cr=0;
        }
        for(int i=0;i<9;i++){
            set<char>st;
            int cr=0;
            for(int j=0;j<9;j++){
                if(board[j][i]!='.')st.insert(board[j][i]),cr++;
            }if(st.size()!=cr)return 0;
            st.clear();cr=0;
        }
        for(int i=0;i<9;i+=3){
            for(int j=0;j<9;j+=3){
                set<char>st;
                int cr=0;
                for(int k=0;k<3;k++){
                    for(int l=0;l<3;l++){
                        if(board[i+k][j+l]!='.')st.insert(board[i+k][j+l]),cr++;
                    }
                }if(st.size()!=cr)return 0;
                st.clear();cr=0;
            }
        }return 1;
    }
};
