class Solution {
public:
    vector<vector<string>> groupAnagrams(vector<string>& strs) {
        map<string,vector<string>>mp;
        for(auto x:strs){
            string s=x;
            sort(s.begin(),s.end());
            mp[s].push_back(x);
        }vector<vector<string>>ans;
        for(auto [x,y]:mp){
            ans.push_back(y);
        }return ans;
    }
};
