class Solution {
public:
    int longestConsecutive(vector<int>& nums) {
        if(nums.empty())return 0;
        set<int>st;
        for(auto x:nums)st.insert(x);
        vector<int>v;
        for(auto x:st)v.emplace_back(x);
        int mx=1,cnt=1;
        for(int i=1;i<st.size();i++){
            if(v[i]==v[i-1]+1)cnt++,mx=max(mx,cnt);
            else cnt=1;
        }return mx;
    }
};
