class Solution {
public:
    vector<int> topKFrequent(vector<int>& nums, int k) {
        map<int,int>mp;
        for(auto x:nums)mp[x]++;
        vector<pair<int,int>>pr;
        for(auto [x,y]:mp){
            pr.emplace_back(y,x);
        }sort(pr.rbegin(),pr.rend());
        vector<int>ans;
        for(int i=0;i<k;i++)ans.push_back(pr[i].second);
        return ans;
    }
};
