## Formula

If `n = p1^a1 · p2^a2 · ... · pk^ak`, then:

```
σ(n) = Π (p_i^(a_i+1) - 1) / (p_i - 1)   for i = 1 to k
```

## Code (C++)

```cpp
#include <bits/stdc++.h>
using namespace std;

long long power(long long base, long long exp) {
    long long result = 1;
    while (exp > 0) {
        if (exp & 1) result *= base;
        base *= base;
        exp >>= 1;
    }
    return result;
}

long long divisor_sum(long long n) {
    long long result = 1;

    for (long long p = 2; p * p <= n; p++) {
        if (n % p == 0) {
            int a = 0;
            while (n % p == 0) {
                n /= p;
                a++;
            }
            long long term = (power(p, a + 1) - 1) / (p - 1);
            result *= term;
        }
    }

    if (n > 1) { // remaining prime factor > sqrt(original n)
        result *= (1 + n);
    }

    return result;
}

int main() {
    long long n;
    cin >> n;
    cout << divisor_sum(n);
}
```

## Code — Sieve Version (for multiple queries)

Precompute the smallest prime factor (SPF) for every number up to N once, then answer σ(n) for any n ≤ N fast using the same formula.

```cpp
#include <bits/stdc++.h>
using namespace std;

vector<int> smallest_prime_factor(int n) {
    vector<int> spf(n + 1);
    iota(spf.begin(), spf.end(), 0);

    for (int i = 2; (long long)i * i <= n; i++) {
        if (spf[i] == i) { // i is prime
            for (int j = i * i; j <= n; j += i) {
                if (spf[j] == j) spf[j] = i;
            }
        }
    }
    return spf;
}

long long power(long long base, long long exp) {
    long long result = 1;
    while (exp > 0) {
        if (exp & 1) result *= base;
        base *= base;
        exp >>= 1;
    }
    return result;
}

long long divisor_sum_spf(int n, vector<int>& spf) {
    long long result = 1;

    while (n != 1) {
        int p = spf[n];
        int a = 0;
        while (n % p == 0) {
            n /= p;
            a++;
        }
        result *= (power(p, a + 1) - 1) / (p - 1);
    }

    return result;
}

int main() {
    int n, q;
    cin >> n >> q; // n = max value to support, q = number of queries

    vector<int> spf = smallest_prime_factor(n);

    while (q--) {
        int x;
        cin >> x;
        cout << divisor_sum_spf(x, spf) << "\n";
    }
}
```

## Complexity

|Step|Complexity|
|---|---|
|Trial division: factorization of n|O(√n)|
|Trial division: total|**O(√n)**|
|SPF sieve: precompute up to N (once)|O(N log log N)|
|SPF sieve: factorize a single query x|O(log x)|
|SPF sieve: Q queries|**O(N log log N + Q log N)**|
