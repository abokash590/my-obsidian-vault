int n,v[6];
vector<vector<int>>dp(4004,vector<int>(4004,-1));
int solve(int i,int need){
    //base
    if(!need)return 0;
    if(i==3)return -1e9;
    if(dp[i][need]!=-1)return dp[i][need];
    int cnt=-1e9;
    //take
    if(v[i]<=need){
        cnt=max(cnt,1+solve(i,need-v[i]));
    }
    //d.t
    cnt=max(cnt,solve(i+1,need));
    return dp[i][need]=cnt;
}
void code(){
    cin>>n>>v[0]>>v[1]>>v[2];
    cout<<solve(0,n);
}
