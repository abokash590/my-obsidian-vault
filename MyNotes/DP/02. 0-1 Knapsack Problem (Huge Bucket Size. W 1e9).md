int n,W,v[105],w[105];
vector<vector<int>>dp(105,vector<int>(1000005,-1));
int solve(int i,int valnow){
    //base
    if(!valnow)return 0;
    if(i==n)return 1e13;
    
    if(dp[i][valnow]!=-1)return dp[i][valnow];

    int res=1e11,op1=LLONG_MAX,op2=LLONG_MAX;
    //take
    if(v[i]<=valnow)op1=w[i]+solve(i+1,valnow-v[i]);
    //dont take
    op2=solve(i+1,valnow);
    res=min(op1,op2);
    dp[i][valnow]=res;
    return res;
} 
void code(){
    cin>>n>>W;
    for(int i=0;i<n;i++)cin>>w[i]>>v[i];
    int ans=0ll;
    // int l=1,r=1e5+4;
    for(int i=1;i<1e5+4;i++){
        if(solve(0,i)<=W){
            if(i>ans)ans=i;
        }
    }cout<<ans<<endl;
}
