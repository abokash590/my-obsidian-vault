int n,x,v[105];
vector<int>dp(1000004,-1);
					int nxt[1000004];
int solve(int need){
    if(!need)return 0;
    if(dp[need]!=-1)return dp[need];
    int c=1e11;
    for(int i=0;i<n;i++){
				      if(v[i]<=need){
				           int x=1+solve(need-v[i]);
				           if(x<c)nxt[need]=need-v[i],c=x;
				      }
    }return dp[need]=c;
}

void code(){
    cin>>n>>x;
    for(int i=0;i<n;i++)cin>>v[i];
    cout<<solve(x)<<endl;
				    while(x){
				        int cur=x-nxt[x];
				        cout<<cur<<' ';
				        x-=cur;
				    }//It will Print The list
}
