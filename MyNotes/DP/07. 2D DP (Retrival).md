int n,W,v[105],w[105];
vector<vector<int>>dp(105,vector<int>(100005,-1));
			vector<vector<int>>nxt(105,vector<int>(100005));
int solve(int i,int wgtnow){
    //base
    if(i==n)return 0;
    if(dp[i][wgtnow]!=-1)return dp[i][wgtnow];
    int res=0ll,op1=0ll,op2=0ll;
    //take
    if(w[i]<=wgtnow){
		        op1=v[i]+solve(i+1,wgtnow-w[i]);
		        if(op1>res)res=op1,nxt[i][wgtnow]=1;
    }
    //dont take
			    op2=solve(i+1,wgtnow);
			    if(op2>res)res=op2,nxt[i][wgtnow]=0;
    return dp[i][wgtnow]=res;
}

void code(){
    cin>>n>>W;
    for(int i=0;i<n;i++)cin>>w[i]>>v[i];
    cout<<solve(0,W)<<endl;
			    for(int i=0;i<n;i++){
			        int cur=nxt[i][W];
			        if(cur)W-=w[i],cout<<v[i]<<' ';
			    }
}
