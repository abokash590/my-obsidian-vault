**Single query (trial division):**

```cpp
#include <bits/stdc++.h>
using namespace std;

int lowest_divisor(int x) {
    if (x % 2 == 0) return 2;
    for (int i = 3; (long long)i * i <= x; i += 2) {
        if (x % i == 0) return i;
    }
    return x; // x is prime, its lowest divisor > 1 is itself
}

int main() {
    int x;
    cin >> x;
    cout << lowest_divisor(x);
}
```

**Multiple queries (precompute with sieve):**


```cpp
#include <bits/stdc++.h>
using namespace std;

vector<int> lowest_divisor_sieve(int n) {
    vector<int> low(n + 1);
    iota(low.begin(), low.end(), 0);

    for (int i = 2; (long long)i * i <= n; i++) {
        if (low[i] == i) { // i is prime
            for (int j = i * i; j <= n; j += i) {
                if (low[j] == j) low[j] = i;
            }
        }
    }
    return low;
}
int main() {
    int n;
    cin >> n;
    vector<int> low = lowest_divisor_sieve(n);
    for (int i = 2; i <= n; i++) {
        cout << i << " -> " << low[i] << "\n";
    }
}
```

## Complexity

|Approach|Complexity|
|---|---|
|Trial division (single x)|O(√x)|
|Sieve precompute (all up to N)|O(N log log N)|
|Sieve query (single x, after precompute)|O(1)|
