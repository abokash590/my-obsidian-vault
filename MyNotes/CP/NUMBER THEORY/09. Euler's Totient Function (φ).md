## Code — Single Number (Prime Factorization)

```cpp
#include <bits/stdc++.h>
using namespace std;
long long phi(long long n) {
    long long result = n;
    for (long long p = 2; p * p <= n; p++) {
        if (n % p == 0) {
            while (n % p == 0) n /= p;
            result -= result / p;
        }
    }
    if (n > 1) result -= result / n;
    return result;
}
int main() {
    long long n;cin >> n;
    cout <<phi(n);
}
```
## Code — Sieve for All Numbers up to N (Eratosthenes-style)

```cpp
#include <bits/stdc++.h>
using namespace std;
vector<int> phi_sieve(int n) {
    vector<int> phi(n + 1);
    iota(phi.begin(), phi.end(), 0);
    for (int p = 2; p <= n; p++) {
        if (phi[p] == p) { // p is prime
            for (int j = p; j <= n; j += p) {
                phi[j] -= phi[j] / p;
            }
        }
    }
    return phi;
}
int main() {
    int n;cin >> n;
    vector<int> phi = phi_sieve(n);
    for (int i = 1; i <= n; i++) {
        cout << i << " -> " << phi[i] << "\n";
    }
}
```

## Complexity

| Approach                                           | Complexity     |
| -------------------------------------------------- | -------------- |
| Single number (trial division factorization)       | O(√n)          |
| Sieve for all numbers up to N (Eratosthenes-style) | O(N log log N) |
|                                                    |                |
