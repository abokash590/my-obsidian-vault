**Harmonic Series Sieve (sum of divisors for all numbers up to N):**

```cpp
#include <bits/stdc++.h>
using namespace std;
vector<long long> divisor_sum_sieve(int n) {
    vector<long long> sum(n + 1, 0);
    for (int i = 1; i <= n; i++) {
        for (int j = i; j <= n; j += i) {
            sum[j] += i;
        }
    }return sum;
}
int main() {
    int n;cin >> n;
    vector<long long> sum = divisor_sum_sieve(n);
    for (int i = 1; i <= n; i++) {
        cout << i << " -> " << sum[i] << "\n";
    }
}
```

**sqrt(N) Approach (sum of divisors for a single number):**

```cpp
#include <bits/stdc++.h>
using namespace std;
long long divisor_sum(long long x) {
    long long sum = 0;
    for (long long i = 1; i * i <= x; i++) {
        if (x % i == 0) {
            sum += i;
            if (i != x / i) sum += x / i;
        }
    }return sum;
}
int main() {
    long long x;cin >> x;
    cout << divisor_sum(x);
}
```
**Used when number is so large. Like ((√n~1e7)**

**It can be solved by block decomposition.**
```cpp
int sm(int n) {
    int a=n%MOD;
    int b=(n+1)%MOD;
    int res=mul(a, b);
    return mul(res,500000004);
}

void code(){
    int n;cin>>n;
    int ans=0ll;
    for(int i=1,r;i<=n;i=r+1){
        int x=n/i;
        r=n/x;
        int sum=sub(sm(r),sm(i-1));
        ans=add(sum*(x%MOD),ans);
    }cout<<ans<<endl;
}
```
## Complexity

| Approach                                                                              | Complexity |
| ------------------------------------------------------------------------------------- | ---------- |
| Harmonic series sieve (all numbers up to N)                                           | O(N log N) |
| Harmonic series sieve, per-number amortized                                           | O(log N)   |
| sqrt(N) approach (single number x)                                                    | O(√x)      |
| <mark style="background: #FFB8EBA6;">Block Decomposition (all numbers up to N)</mark> | O(√x)      |
