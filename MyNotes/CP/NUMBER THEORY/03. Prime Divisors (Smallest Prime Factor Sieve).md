## Code (C++)

```cpp
#include <bits/stdc++.h>
using namespace std;

vector<int> smallest_prime_factor(int n) {
    vector<int> spf(n + 1);
    iota(spf.begin(), spf.end(), 0);

    for (int i = 2; (long long)i * i <= n; i++) {
        if (spf[i] == i) { // i is prime
            for (int j = i * i; j <= n; j += i) {
                if (spf[j] == j) spf[j] = i;
            }
        }
    }
    return spf;
}
vector<int> prime_divisors(int x, vector<int>& spf) {
    vector<int> primes;
    while (x != 1) {
        int p = spf[x];
        primes.push_back(p);
        while (x % p == 0) x /= p;
    }
    return primes;
}
int main() {
    int n;
    cin >> n;
    vector<int> spf = smallest_prime_factor(n);
    vector<int> primes = prime_divisors(n, spf);
    for (int p : primes) cout << p << " ";
}
```

## Complexity

|Operation|Complexity|
|---|---|
|Build SPF sieve (once, up to N)|O(N log log N)|
|Get prime divisors of a single x (using SPF)|O(log x)|
|Get prime divisors of x (without sieve, trial division)|O(√x)|
