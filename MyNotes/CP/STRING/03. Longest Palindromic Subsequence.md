**Make n^2 from m^3 by using DP**

```cpp
string s;
vector<vector<int>>dp(1001,vector<int>(1001,-1));
int isPal(int i,int j){
    if(i>j)return 0;
    if(i==j)return 1;
    if(dp[i][j]!=-1)return dp[i][j];
    int ans=0ll;
    if(s[i]==s[j])ans=max(ans,2ll+isPal(i+1,j-1));
    ans=max(ans,isPal(i+1,j));
    ans=max(ans,isPal(i,j-1));
    return dp[i][j]=ans;
}
```

**Iterative Version:**
```cpp
void code(){
    string s;cin>>s;
    int q;cin>>q;
    int n=s.size();
    vector<vector<int>>dp(n,vector<int>(n));
    for(int i=0;i<n;i++)dp[i][i]=1;
    for(int len=2;len<=n;len++){
        for(int i=0;i+len-1<n;i++){
            int j=i+len-1;
            if(s[i]==s[j])dp[i][j]=((len==2)?2ll:2ll+dp[i+1][j-1]);
            else dp[i][j]=max(dp[i][j-1],dp[i+1][j]);
        }
    }
    while(q--){
        int l,r;cin>>l>>r;
        cout<<dp[--l][--r]<<endl;
    }
}
```
