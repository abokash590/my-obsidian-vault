**Make n^2 from m^3 by using DP**

```cpp
string s;
vector<vector<int>>dp(1001,vector<int>(1001,-1));
bool isPal(int i,int j){
    if(i>=j)return 1;
    if(dp[i][j]!=-1)return dp[i][j];
    if(s[i]!=s[j])return dp[i][j]=0;
    return dp[i][j]=isPal(i+1,j-1);
}

void code(){
    cin>>s;
    int q;cin>>q;
    int n=s.size();
    while(q--){
        int l,r;cin>>l>>r;
        cout<<isPal(--l,--r)<<endl;
    }
}
```

**Iterative Version:**
```cpp
void code(){
    string s;cin>>s;
    int q;cin>>q;
    int n=s.size();
    vector<vector<int>>dp(n,vector<int>(n));
    for(int i=0;i<n;i++)dp[i][i]=1;
    for(int len=2;len<=n;len++){
        for(int i=0;i+len-1<n;i++){
            int j=i+len-1;
            if(s[i]==s[j])dp[i][j]=((len==2)?1:dp[i+1][j-1]);
        }
    }
    while(q--){
        int l,r;cin>>l>>r;
        cout<<dp[--l][--r]<<endl;
    }
}
```
